            if (m) return parseInt(m[1], 10) - 1;
            node = node.parentElement;
        }
        return -1;
    };

    const canvasInkRatio = (canvas) => {
        if (!canvas?.width || !canvas?.height) return 0;
        try {
            const { width: w, height: h } = canvas;
            const data = canvas.getContext('2d').getImageData(0, 0, w, h).data;
            let ink = 0;
            let total = 0;
            const step = Math.max(1, Math.floor(w * h / 120000));
            for (let i = 0; i < data.length; i += 4 * step) {
                total++;
                if (data[i] < 240 || data[i + 1] < 240 || data[i + 2] < 240) ink++;
            }
            return total ? ink / total : 0;
        } catch (e) {
            return 0;
        }
    };

    const parseApiResponse = (respText) => {
        const t = (respText || '').trim();
        if (!t) throw new Error('空响应');
        try {
            return JSON.parse(t);
        } catch (e) { /* jsonp / js */ }
        const p = t.indexOf('(');
        const q = t.lastIndexOf(')');
        if (p >= 0 && q > p) {
            return JSON.parse(t.substring(p + 1, q));
        }
        const eq = t.match(/=\s*(\{[\s\S]*\})\s*;?\s*$/);
        if (eq) return JSON.parse(eq[1]);
        throw new Error('无法解析响应: ' + t.slice(0, 120));

    const parseJsonPageToParagraphs = (resultData) => {
        let data = resultData;
        if (data && typeof data === 'object' && !Array.isArray(data) && !data.body && !data.parags) {
            if (Array.isArray(data.data)) data = data.data;
            else if (Array.isArray(data.page)) data = data.page;
            else if (data.pages) data = data.pages;
        }
        const paragraphs = [];

        const flushLine = (line) => {
            const t = (line || '').trim();
            if (!t) {
                paragraphs.push(BLANK_PARA);
                return;
            }
            splitDenseParagraph(t).forEach(p => paragraphs.push(p));
        };

        const walkParags = (parags) => {
            if (!parags?.length) return;
            let line = '';
            parags.forEach(p => {
                if (!p || typeof p.c !== 'string') return;
                const c = p.c;
                if (c === '\n' || c === '\r\n' || p.ps === 1 || p.ps === '1') {
                    flushLine(line);
                    line = '';
                } else {
                    line += c;
                }
            });
            flushLine(line);
        };

        if (Array.isArray(data)) {
            data.forEach(block => {
                if (block?.parags) walkParags(block.parags);
                else if (typeof block?.c === 'string') flushLine(block.c);
            });
        } else if (data?.parags) {
            walkParags(data.parags);
        } else if (data?.body) {
            let line = '';
            data.body.forEach(item => {
                if (typeof item.c !== 'string') return;
                if (item.c === '\n' || item.ps === 1 || item.ps === '1') {
                    flushLine(line);
                    line = '';
                } else {
                    line += item.c;
                }
            });
            flushLine(line);
        } else if (typeof data?.c === 'string') {
            flushLine(data.c);
        }
        return paragraphs;
    };

    /** 保留连续换行（空行）；无换行符时长段按题号切分 */
    const splitDenseParagraph = (text) => {
        if (!text) return [];
        if (text.includes('\n')) {
            const out = [];
            text.split('\n').forEach(s => {
                const t = s.trim();
                if (t) {
                    if (!t.includes('\n') && t.length > 120) {
                        const byNum = t.split(/(?=（\d+）)|(?=\(\d+\))|(?=\d+[、.．]\s*)/)
                            .map(x => x.trim()).filter(Boolean);
                        if (byNum.length > 1) byNum.forEach(x => out.push(x));
                        else out.push(t);
                    } else {
                        out.push(t);
                    }
                } else {
                    out.push(BLANK_PARA);
                }
            });
            return out;
        }
        const byNum = text.split(/(?=（\d+）)|(?=\(\d+\))|(?=\d+[、.．]\s*)/).map(s => s.trim()).filter(Boolean);
        if (byNum.length > 1) return byNum;
        return [text];
    };

            c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
            resolve(canvasToPngDataUrl(c));
        };

        img.src = dataUrl;
    });

    /** 从页图按空白带切出插图（百度文库 / 道客巴巴共用） */
    const extractFigureRegionsFromCanvas = (source) => {
        if (!source?.width || !source?.height) return [];
        const w = source.width;
        const h = source.height;
        let data;
        try {
            data = source.getContext('2d').getImageData(0, 0, w, h).data;
        } catch (e) {
            return [];
        }
        const rowActive = (y) => {
            let n = 0;
            const step = Math.max(1, Math.floor(w / 80));
            for (let x = 0; x < w; x += step) {
                const i = (y * w + x) * 4;
                if (data[i] < 245 || data[i + 1] < 245 || data[i + 2] < 245) n++;
            }
            return n / Math.ceil(w / step) > 0.028;
        };
        const bands = [];
        let start = -1;
        for (let y = 0; y < h; y++) {
            if (rowActive(y)) {
                if (start < 0) start = y;
            } else if (start >= 0) {
                if (y - start >= 24) bands.push({ top: start, height: y - start });
                start = -1;
            }
        }
        if (start >= 0 && h - start >= 24) bands.push({ top: start, height: h - start });
        const merged = [];
        bands.forEach(b => {
            const last = merged[merged.length - 1];
            if (last && b.top - (last.top + last.height) < 14) {
                last.height = b.top + b.height - last.top;
            } else {
                merged.push({ ...b });
            }
        });
        const minH = Math.max(40, Math.round(h * 0.06));
        const maxH = Math.round(h * 0.72);
        const figures = merged.filter(b => b.height >= minH && b.height <= maxH && (b.height / h) >= 0.06);
        const out = [];
        for (const g of figures) {
            const c = u.createEl('canvas');
            c.width = w;
            c.height = g.height;
            const ctx = c.getContext('2d');
            ctx.fillStyle = '#FFFFFF';
            ctx.fillRect(0, 0, w, g.height);
            ctx.drawImage(source, 0, g.top, w, g.height, 0, 0, w, g.height);
            if (canvasInkRatio(c) < 0.003) continue;
            const dataUrl = canvasToPngDataUrl(c);
            if (!isValidImageDataUrl(dataUrl)) continue;
            out.push({
                dataUrl,
                width: Math.min(340, Math.round(w * 0.72)),
                heightPx: g.height,
                top: g.top
            });
        }
        return out;
    };

    const expandParagraphsForOptions = (paragraphs) => {
        const out = [];
        (paragraphs || []).forEach(p => {
            if (isBlankPara(p)) {
                out.push(BLANK_PARA);
                return;

            const parts = String(p).split(/(?=\s[A-Da-d][.．、:：])/);
            if (parts.length > 1) {
                parts.forEach(s => {
                    const t = s.trim();
                    if (t) out.push(t);

                out.push(p);

        });

    const splitDoc88QuestionGroups = (paragraphs) => {
        const groups = [];
        let cur = { lines: [] };
        (paragraphs || []).forEach(p => {
            if (isBlankPara(p)) {
                if (cur.lines.length) groups.push(cur);
                groups.push({ lines: [BLANK_PARA], blankOnly: true });
                cur = { lines: [] };
                return;
            }
            if (DOC88_QUESTION_HEAD.test(p) && cur.lines.length) {
                groups.push(cur);
                cur = { lines: [p] };
            } else if (DOC88_QUESTION_HEAD.test(p)) {
                cur = { lines: [p] };
            } else {
                cur.lines.push(p);
            }
        });
        if (cur.lines.length) groups.push(cur);
        return groups;
    };

        const images = JSON.parse(GM_getValue('listData') || '[]');
        const text = images.map(item => item.src).join('\n');
        downloadBlob(new Blob([text], { type: 'text/plain;charset=utf-8' }), sanitizeTitle(title) + '_urls.txt');
    };

    const downpdf = () => pdfDoc.save(sanitizeTitle(title) + '.pdf', { returnPromise: true });

    const downzip = () => {
        zipWriter.close().then(blob => downloadBlob(blob, sanitizeTitle(title) + '.zip'))
            .catch(e => console.error(e));
    };

    const conditionDownload = () => {
        if (downType === 1) {
            downpdf();
            localStorage.setItem('down', '1');
        } else if (downType === 2) {
            downzip();
            localStorage.setItem('down', '1');
        }
        u.update('text', '下载完成');
    };

    const parseImage = async () => {
        const images = u.queryAll(select);
        if (!images.length) {
            alert('暂无画布数据，请先自动预览');
            return;
        }
        const listData = [];
        images.forEach(item => {
            let src = u.attr(item, 'data-src') || item.src;
            if (!src) return;
            if (src.startsWith('//')) src = 'https:' + src;
            if (src.startsWith('http:') && location.protocol === 'https:') src = src.replace('http:', 'https:');
            listData.push({ src, page: listData.length });
        });
        const store = JSON.stringify(listData);
        GM_setValue('listData', store);
        localStorage.setItem('listData', store);
        localStorage.removeItem('current');
        await downimg();
    };

    const parentMessage = (src, frameId) => {
        const frame = document.getElementById(frameId);
        frame?.contentWindow?.postMessage({ type: 'parent', value: src }, '*');
    };

    const downimg = async () => {
        const images = JSON.parse(GM_getValue('listData') || '[]');
        const length = images.length;
        if (!length) {
            alert('没有图片地址，请等待文档加载或先自动预览');
            return;
        }
        localStorage.setItem('length', String(length));
        let current = Number(localStorage.getItem('current') || '0');

        for (let index = current; index < length; index++) {
            const src = images[index].src;
            const srcHost = new URL(src, location.href).host;
            const sameOrigin = src.includes('baidu.com') || src.includes(host);

            if (!sameOrigin) {
                const frameId = 'frame_' + srcHost.replace(/\./g, '_');
                let frame = document.getElementById(frameId);
                if (!frame) {
                    frame = u.createEl('iframe', frameId);
                    frame.src = new URL(src).origin + '?custom=true';
                    frame.style.cssText = 'visibility:hidden;width:0;height:0;border:0';
                    document.body.appendChild(frame);
                }
                localStorage.setItem('current', String(index));
                await u.preview(index, length);
                parentMessage(src, frameId);
                return;
            }

            try {
                const { uint8, blob, width, height } = await MF_ImageToBase64(src);
                if (fileType.includes('ppt') || width > height) {
                    pdfDoc.addPage([width * pdfRatio, height * pdfRatio], 'l');
                    pdfDoc.addImage(uint8, 'JPEG', 0, 0, width * pdfRatio, height * pdfRatio, index, 'FAST');
                } else {
                    pdfDoc.addPage();
                    pdfDoc.addImage(uint8, 'JPEG', 0, 0, pdfW, pdfH, index, 'FAST');
                }
                if (index === 1) pdfDoc.deletePage(1);
                zipWriter.add(index + '.png', new zip.BlobReader(blob));
                await u.preview(index + 1, length);
            } catch (e) {
                console.warn('图片失败', src, e);
            }
            current = index;
        }

        if (localStorage.getItem('current')) return;
        if (current === length - 1) conditionDownload();
    };

    const scrollWinArea = () => {
        before();
        const scrollTop = dom.scrollTop;
        const height = dom.scrollHeight - dom.clientHeight;
        if (height <= scrollTop + 5) {
            stopPreview();
            u.preview(-1);
            return;
        }
        dom.scrollTo({ top: scrollTop + 500, left: 0, behavior: 'smooth' });
        u.preview(scrollTop, height);
        accumulateVisiblePages();
    };

    /** 按页滚动并缓存（避免虚拟列表只保留 2 页 DOM） */
    const scrollPreviewBaidu = async () => {
        if (!localStorage.getItem('start')) {
            u.preview(-1, null, '已终止');
            return;
        }
        before();
        const i = Number(localStorage.getItem('SP_index') || '0');
        const childrens = u.queryAll(select);
        const current = childrens[i];
        const length = childrens.length;
        if (!current) {
            stopPreview();
            u.preview(-1);
            return;
        }
        const pageHost = current.parentElement || current;
        let canvas = current.tagName === 'CANVAS' ? current : pageHost.querySelector('canvas');
        if (!canvas) canvas = pageHost.querySelector('.canvasWrapper canvas');
        const textLayer = pageHost.querySelector('.textLayer') ||
            current.closest('#original-creader-root')?.querySelector('.textLayer');
        const dataLoaded = pageHost.getAttribute?.('data-loaded');
        if ((dataLoaded || isElementInViewport(pageHost)) && (canvas || textLayer)) {
            accumulatePageAtIndex(i, textLayer, canvas);
            if (i < length - 1) {
                const nextHost = childrens[i + 1].parentElement || childrens[i + 1];
                nextHost?.scrollIntoView({ behavior: 'smooth', block: 'center' });
            }
        } else {
            pageHost.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }
        await u.preview(i + 1, length);
        if (i < length - 1) {
            localStorage.setItem('SP_index', String(i + 1));
            setTimeout(scrollPreviewBaidu, 500);
        } else {
            u.preview(-1);
            localStorage.removeItem('SP_index');
            localStorage.removeItem('start');
            console.log('[百度文库] 预览完成, 已缓存', Object.keys(loadWordPageCache()).length, '页');
        }
    };

    const cacheVisibleTextLayers = () => {
        accumulateVisiblePages();
    };

    const stopPreview = () => {
        if (interval) {
            clearInterval(interval);
            interval = null;
        }
        localStorage.removeItem('start');
    };

    const autoPreview = async () => {
        if (interval) return;
        localStorage.setItem('start', '1');
        localStorage.removeItem('SP_index');
        dom.scrollTop = 0;
        before();
        if (isDoc88) {
            const pages = getDoc88PageElements().length;
            if (pages > 0) {
                await scrollPreviewDoc88Tick();
            } else {
                interval = setInterval(scrollPageAreaDoc88, 500);
            }
            return;
        }
        const canvasCount = u.queryAll(select).length;
        if (canvasCount > 0 && (fileType.includes('word') || fileType.includes('doc'))) {
            await scrollPreviewBaidu();
            return;
        }
        interval = setInterval(scrollWinArea, 500);
    };

    const executeDownload = async (type) => {
        downType = type;
        if (localStorage.getItem('down') && type !== 3) {
            conditionDownload();
            return;
        }
        u.update('text', '准备下载…');
        try {
            if (type === 3) {
                if (isDoc88) {
                    syncDoc88CoreTextsToCache();
                    const pages = await collectDoc88WordContent();
                    const ok = await exportWordRich(pages);
                    u.update('text', ok ? 'Word 下载完成' : 'Word 导出失败');
                } else {
                    syncPageDataFromReader();
                    const pages = await collectBaiduWordContent();
                    const ok = await exportWordRich(pages);
                    u.update('text', ok ? 'Word 下载完成' : 'Word 导出失败');
                }
                return;
            }
            if (isDoc88) {
                await parseImageDoc88();
                return;
            }
            resetWriters();
            const display = u.query('#app-top-right-tool')?.style.display;
            if (!display || display === 'none') {
                await parseImage();
            } else {
                conditionDownload();
            }
        } catch (err) {
            console.error(err);
            alert('下载失败：' + (err.message || err));
            u.update('text', '下载失败');
        }
    };

    const handleRenderData = () => {
        const pd = unsafeWindow.pageData;
        if (!pd?.readerInfo) return;
        const { readerInfo, viewBiz } = pd;
        fileType = (viewBiz?.docInfo?.fileType || fileType).toLowerCase();
        title = sanitizeTitle(viewBiz?.docInfo?.aiQueryTitle || title);
        readerInfoBai = readerInfo;
        const htmlUrls = readerInfo.htmlUrls;
        if (htmlUrls) {
            if (fileType.includes('ppt')) {
                MF_addURL(htmlUrls);
            } else if (fileType.includes('pdf')) {
                let images = htmlUrls.png
                    ? htmlUrls.png.map(item => item.pageLoadUrl)
                    : htmlUrls;
                MF_addURL(images);
            }
            if (htmlUrls.json) {
                MF_addURL(htmlUrls.json.map(item => item.pageLoadUrl), 'pageData');
            }
            if (htmlUrls.png?.length) {
                MF_addURL(htmlUrls.png.map(item => item.pageLoadUrl));
            }
        }
        if (fileType === 'txt' && readerInfoBai) {
            const { docId, freePage, rsign, showPage, md5sum } = readerInfoBai;
            const urls = [];
            for (let i = 1; i < showPage + 1; i++) {
                const x = md5sum.substring(1);
                const n = ['pn=' + i, 'rn=1', 'type=txt', 'spr=0', 'rsign=' + rsign, 'callback=wenku_' + i].join('&');
                urls.push(BASE_URL + '/text/' + docId + '?' + x + '&' + n);
            }
            MF_addURL(urls, 'pageData');
        }
    };

    const hookBaiduXhr = () => {
        const _open = unsafeWindow.XMLHttpRequest.prototype.open;
        unsafeWindow.XMLHttpRequest.prototype.open = function (...args) {
            const reqUrl = String(args[1] || '');
            this.addEventListener('load', () => {
                try {
                    capturePageUrlFromXhr(reqUrl);
                    const url = reqUrl;
                    if (!url.includes('ndocview/readerinfo')) return;
                    const { data } = JSON.parse(this.responseText);
                    readerInfoBai = data;
                    if (data.fileType) fileType = String(data.fileType).toLowerCase();
                    const htmlUrls = data.htmlUrls;
                    if (!htmlUrls) return;
                    let images = htmlUrls.png
                        ? htmlUrls.png.map(item => item.pageLoadUrl)
                        : htmlUrls;
                    if (htmlUrls.json) {
                        MF_addURL(htmlUrls.json.map(item => item.pageLoadUrl), 'pageData');
                    }
                    if (fileType === 'txt') {
                        const { docId, freePage, rsign, md5sum } = data;
                        const urls = [];
                        const start = (readerInfoBai?.showPage || 0) + 1;
                        for (let i = start; i < freePage + 1; i++) {
                            const x = md5sum.substring(1);
                            const n = ['pn=' + i, 'rn=1', 'type=txt', 'spr=0', 'rsign=' + rsign, 'callback=wenku_' + i].join('&');
                            urls.push(BASE_URL + '/text/' + docId + '?' + x + '&' + n);
                        }
                        MF_addURL(urls, 'pageData');
                    }
                    MF_addURL(images);
                } catch (e) { /* ignore */ }
            });
            return _open.apply(this, args);
        };
    };

    const initDoc88 = () => {
        if (!/doc88\.com\/.+/.test(href)) return;
        dom = document.documentElement;
        title = sanitizeTitle(u.query('meta[property="og:title"]')?.content || document.title);
        const attrLine = u.query('#box1 .attribute')?.innerText || '';
        const typePart = attrLine.split(/[：:]/).pop();
        fileType = (typePart || 'doc').trim().toLowerCase();
        beforeFun = "let eb=u.query('#continueButton');if(eb)eb.click();";
        select = '#pageContainer canvas[id^="page_"]';
        enableDoc88Copy();
        hookDoc88Xhr();
        hookDoc88CopyCapture();
        hookDoc88TextArray();

        const buttons = [
            { id: 'text', label: '状态 0%' },
            { id: 'start', label: '自动预览' },
            { id: 'stop', label: '停止预览' },
            { id: 'down', label: '下载图片' },
            { id: 'pdf', label: '下载PDF' },
            { id: 'word', label: '下载Word' },
            { id: 'txt', label: '导出TXT' },
            { id: 'copy', label: '复制全文' }
        ];
        const handlers = {
            start: () => autoPreview(),
            stop: () => stopPreview(),
            down: () => executeDownload(2),
            pdf: () => executeDownload(1),
            word: () => executeDownload(3),
            txt: () => fullText(),
            copy: () => copyFullTextToClipboard()
        };
        u.gui(buttons, handlers);
        ['copy', 'selectstart', 'contextmenu'].forEach(evt => {
            document.addEventListener(evt, (e) => e.stopImmediatePropagation(), true);
        });
        console.log('[道客巴巴] 标题:', title, '类型:', fileType);
    };

    const initBaidu = () => {
        if (!/wenku\.baidu\.com\/(tfview|view)/.test(href)) return;
        dom = u.query('.reader-wrap') || document.documentElement;
        handleRenderData();

        const buttons = [
            { id: 'text', label: '状态 0%' },
            { id: 'start', label: '自动预览' },
            { id: 'stop', label: '停止预览' },
            { id: 'down', label: '下载图片' },
            { id: 'pdf', label: '下载PDF' },
            { id: 'word', label: '下载Word' },
            { id: 'txt', label: '导出TXT' },
            { id: 'copy', label: '复制全文' }
        ];
        const handlers = {
            start: () => autoPreview(),
            stop: () => stopPreview(),
            down: () => executeDownload(2),
            pdf: () => executeDownload(1),
            word: () => executeDownload(3),
            txt: () => fullText(),
            copy: () => copyFullTextToClipboard()
        };
        if (fileType.includes('ppt') || fileType.includes('pdf')) {
            buttons.push({ id: 'urls', label: '获取地址' });
            handlers.urls = () => downtxt();
        }
        buttons.push({ id: 'print', label: '打印PDF' });
        handlers.print = () => window.print();

        u.gui(buttons, handlers);
        console.log('[百度文库] 标题:', title, '类型:', fileType);
    };

    // 子页面：bdimg CDN 拉图（仅百度）
    if (isChild && params.get('custom')) {
        window.parent.postMessage({ type: 'onload', value: 'success' }, '*');
        window.addEventListener('message', async (e) => {
            if (e.data?.type === 'parent' && e.data.value) {
                try {
                    const data = await MF_ImageToBase64(e.data.value);
                    window.parent.postMessage({ type: 'child', value: data }, '*');
                } catch (err) {
                    console.error(err);
                }
            }
        });
        return;
    }

    if (isBaidu && host.includes('wenku.baidu.com')) {
        hookBaiduXhr();
        clearCache();
        if (href.includes('tfview') || href.includes('view')) {
            GM_deleteValue('listData');
            setTimeout(handleRenderData, 800);
        }
        setTimeout(initBaidu, 1200);
    }

    if (isDoc88) {
        clearCache();
        enableDoc88Copy();
        hookDoc88Xhr();
        hookDoc88CopyCapture();
        setTimeout(() => {
            enableDoc88Copy();
            hookDoc88CopyCapture();
            hookDoc88TextArray();
            deepScanCoreForPageTexts();
            initDoc88();
        }, 1200);
        setInterval(() => {
            enableDoc88Copy();
            hookDoc88CopyCapture();
            hookDoc88TextArray();
        }, 3000);
    }

    window.addEventListener('message', async (e) => {
        if (!isBaidu) return;
        if (e.data?.type !== 'child' || !e.data.value) return;
        const index = Number(localStorage.getItem('current') || '0');
        const { blob, uint8, width, height } = e.data.value;
        if (width > height) {
            pdfDoc.addPage([width * pdfRatio, height * pdfRatio], 'l');
            pdfDoc.addImage(uint8, 'JPEG', 0, 0, width * pdfRatio, height * pdfRatio, index, 'FAST');
        } else {
            pdfDoc.addPage();
            pdfDoc.addImage(uint8, 'JPEG', 0, 0, pdfW, pdfH, index, 'FAST');
        }
        if (index === 1) pdfDoc.deletePage(1);
        zipWriter.add(index + '.png', new zip.BlobReader(blob));
        localStorage.setItem('current', String(index + 1));
        downimg();
    });
})();
